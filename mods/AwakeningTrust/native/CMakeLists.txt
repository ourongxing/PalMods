cmake_minimum_required(VERSION 3.22)
project(AwakeningTrust LANGUAGES CXX)
set(CMAKE_CONFIGURATION_TYPES Game__Shipping__Win64 CACHE STRING "SDK configuration" FORCE)
set(CMAKE_CXX_FLAGS_GAME__SHIPPING__WIN64 "${CMAKE_CXX_FLAGS_RELEASE}")
set(CMAKE_SHARED_LINKER_FLAGS_GAME__SHIPPING__WIN64 "${CMAKE_SHARED_LINKER_FLAGS_RELEASE}")
set(PalModsSDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.build/UE4SS" CACHE PATH "Shared SDK")
find_package(PalModsSDK CONFIG REQUIRED)
add_library(AwakeningTrustNative SHARED src/main.cpp)
option(AWAKENINGTRUST_VALIDATION "Read-only native interface validation on local save copies" OFF)
if(AWAKENINGTRUST_VALIDATION)
    target_compile_definitions(AwakeningTrustNative PRIVATE AWAKENINGTRUST_VALIDATION)
endif()
target_compile_features(AwakeningTrustNative PRIVATE cxx_std_23)
target_include_directories(AwakeningTrustNative PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(AwakeningTrustNative PRIVATE PalMods::UE4SS)
set_property(TARGET AwakeningTrustNative PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_options(AwakeningTrustNative PRIVATE /EHsc /Zc:__cplusplus /utf-8)
set_target_properties(AwakeningTrustNative PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/bin")
add_executable(AwakeningTrustRulesTest src/rules_test.cpp)
target_compile_features(AwakeningTrustRulesTest PRIVATE cxx_std_23)
set_target_properties(AwakeningTrustRulesTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/bin")
