cmake_minimum_required(VERSION 3.22)
project(BetterStorage LANGUAGES CXX)
set(CMAKE_CONFIGURATION_TYPES Game__Shipping__Win64 CACHE STRING "SDK configuration" FORCE)
set(CMAKE_CXX_FLAGS_GAME__SHIPPING__WIN64 "${CMAKE_CXX_FLAGS_RELEASE}")
set(CMAKE_SHARED_LINKER_FLAGS_GAME__SHIPPING__WIN64 "${CMAKE_SHARED_LINKER_FLAGS_RELEASE}")
set(PalModsSDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.build/UE4SS" CACHE PATH "Shared SDK")
find_package(PalModsSDK CONFIG REQUIRED)
add_library(BetterStorageNative SHARED src/main.cpp)

target_compile_features(BetterStorageNative PRIVATE cxx_std_23)
target_include_directories(BetterStorageNative PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(BetterStorageNative PRIVATE PalMods::UE4SS)
set_property(TARGET BetterStorageNative PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_options(BetterStorageNative PRIVATE /EHsc /Zc:__cplusplus /utf-8)
set_target_properties(BetterStorageNative PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/bin")
